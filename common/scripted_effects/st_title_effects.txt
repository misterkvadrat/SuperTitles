st_create_high_kingdom_effect = {
	primary_title = {
		save_scope_as = st_old_primary_title
	}

	create_dynamic_title = {
		tier = hegemony
		name = $TITLE_NAME$
		adj = $TITLE_ADJ$
	}

	scope:new_title = {
		save_scope_as = st_new_title
		set_color_from_title = scope:st_old_primary_title
		set_capital_county = root.capital_county
		set_definitive_form = yes
		set_can_be_named_after_dynasty = no
		set_can_use_nomadic_naming = no
		set_no_automatic_claims = yes
		generate_coa = yes
	}

	create_title_and_vassal_change = {
		type = created
		save_scope_as = st_title_change
		add_claim_on_loss = no
	}

	scope:st_new_title = {
		change_title_holder = {
			holder = root
			change = scope:st_title_change
		}
	}

	resolve_title_and_vassal_change = scope:st_title_change
	set_primary_title_to = scope:st_new_title
}
